#!/bin/bash
# This script is written by Sergiy Bubin. It sets up environment and installs essential packages in
# a fresh Ubuntu 24.04 (WSL) installation. It is intended to be run as a regular user with sudo
# privileges. Note that this script may take a long time to complete due to the large number of 
# packages being installed. If necessary, it can be interrupted, restarted, and run multiple times.

# Note that if you re-install WSL/Ubuntu 24.04 rather than having a first-time install, you need 
# to make sure that you remove WSL/Ubuntu 24.04 properly. To do so, open Windows PowerShell with
# admin privileges. Then see the list of distros by typing
#
# wsl --list --verbose
#
# You will see something like this (e.g. if you have two different versions):
#
# NAME            STATE           VERSION
# Ubuntu-24.04    Running         2
# Ubuntu-22.04    Stopped         2
#
# Then shutdown WSL and unregister/delete Ubuntu 24.04 (or whatever distro you want to reinstall):
#
# wsl --shutdown
# wsl --unregister Ubuntu-24.04
#
# Then remove/uninstall Ubuntu 24.04 from Windows Apps

#To install Ubuntu 24.04 in cmd you can use the following command (assuming
#that the WSL feature has been enabled in Windows):
# wsl --install -d Ubuntu-24.04


set -e

# Check if RSA public key for OpenSSH is available in ~/.ssh/id_rsa.pub and generate one if not
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Checking for SSH RSA public key and setting up SSH config file..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
mkdir -p "$HOME/.ssh"
chmod 700 "$HOME/.ssh"
if [ ! -f ~/.ssh/id_rsa.pub ]; then
    echo "SSH RSA public key not found. Generating one..."
    ssh-keygen -t rsa -b 4096 -f ~/.ssh/id_rsa -N ""
else
    echo "SSH RSA public key found. No need to generate."
fi
# Setting up SSH config file
SSH_CONFIG_FILE="$HOME/.ssh/config"

# Create config file ~/.ssh/config if it does not exist
if [ ! -f "$SSH_CONFIG_FILE" ]; then
    touch "$SSH_CONFIG_FILE"
    chmod 600 "$SSH_CONFIG_FILE"
fi
# Check if Host * with ServerAliveInterval already set in ~/.ssh/config
if ! awk '
  /^\s*Host\s+\*/ {in_host=1; next}
  /^\s*Host\s+/   {in_host=0}
  in_host && /^\s*ServerAliveInterval\s+240(\s|$)/ {found=1}
  END { exit (found ? 0 : 1) }
' "$SSH_CONFIG_FILE"; then
  cat >> "$SSH_CONFIG_FILE" <<'EOF'

Host *
    ServerAliveInterval 240
EOF
fi

# Update and upgrade the system
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Updating and upgrading the system..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
sudo apt update && sudo apt upgrade -y

# Install essential packages
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Installing many essential packages using apt-get..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
sudo apt-get install -y nano csh tcsh zsh rpm gcc gfortran g++ gdb make cmake git gh liblapack-dev liblapacke-dev libopenblas-dev libfftw3-dev libssl-dev python3 python3-all-dev python3-pip python3-numpy python3-scipy python3-matplotlib build-essential wget curl openmpi-bin libopenmpi-dev genius octave gnuplot texlive-full ffmpeg qpdf default-jdk lmod patch rsync file tar gzip bzip2 xz-utils unzip lua-posix tcl python-is-python3 texinfo bison flex libgmp-dev libmpfr-dev libmpc-dev gawk libc6-dev linux-libc-dev

# Install matching system venv package (in Ubuntu 24.04 that is python3.12-venv)
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Installing Python venv package..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
PYTHON_VERSION=$(python3 -c "import sys; print(f'{sys.version_info.major}.{sys.version_info.minor}')")
VENV_PACKAGE="python${PYTHON_VERSION}-venv"
sudo apt-get install -y "$VENV_PACKAGE"

# Some useful settings
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Setting up wsl.conf..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
# Configure WSL to not auto-generate /etc/hosts to preserve customizations
WSL_CONF_FILE="/etc/wsl.conf"
# Ensure the file exists
sudo touch "$WSL_CONF_FILE"
# Check if [network] section exists already in wsl.conf
if ! grep -q '^\[network\]' "$WSL_CONF_FILE"; then
    # Add the section and the setting at the end
    sudo tee -a "$WSL_CONF_FILE" > /dev/null <<'EOF'

[network]
generateHosts = false
EOF
else
    # [network] exists — check if generateHosts is already set
    if ! awk '
      /^\[network\]/ {in_section=1; next}
      /^\[/          {in_section=0}
      in_section && /^[[:space:]]*generateHosts[[:space:]]*=[[:space:]]*false([[:space:]]|$)/ {found=1}
      END { exit (found ? 0 : 1) }
    ' "$WSL_CONF_FILE"; then
      sudo sed -i '/^\[network\]/a generateHosts = false' "$WSL_CONF_FILE"
    fi
fi

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Setting nano as the default text editor ..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
# Make `nano` the default text editor if it is not yet (it adds a line to /etc/bash.bashrc if it is missing) 
sudo sed -i '\|^export EDITOR=nano$|!{ $a\
export EDITOR=nano
}' /etc/bash.bashrc


echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Creating directories for Easybuild and modules..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
# Create directory `/shared` and `/shared/opt` to replicate the paths and environment in NU computer clusters (Muon, Shabyt, Irgetas, etc).
# If necessary, any other location can be used by defining MY_SOFTWARE_INST_DIR variable below.
MY_SOFTWARE_INST_DIR="/shared/opt"
sudo mkdir -p "${MY_SOFTWARE_INST_DIR}"
# Create directory /shared/opt/modules for non-Easybuild modules if they ever created later (Easybuild will have its own modules directory).
# IF necessary, any other location can be used by defining MY_NON_EB_MOD_DIR variable below.
MY_NON_EB_MOD_DIR="/shared/opt/modules"
sudo mkdir -p "${MY_NON_EB_MOD_DIR}"

echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Installing and configuring EasyBuild in ${MY_SOFTWARE_INST_DIR}/easybuild..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"

EASYBUILD_PREFIX="${MY_SOFTWARE_INST_DIR}/easybuild"
EASYBUILD_VENV="${EASYBUILD_PREFIX}/venv"
EASYBUILD_CFG="${EASYBUILD_PREFIX}/config.cfg"
EASYBUILD_EASYCONFIGS_REPO="${EASYBUILD_PREFIX}/easyconfigs"

# Ensure target directories exist (and are writable by the current user)
sudo mkdir -p "${EASYBUILD_PREFIX}"
sudo chown -R "${USER}:${USER}" "${MY_SOFTWARE_INST_DIR}" "${EASYBUILD_PREFIX}"

# Create Python virtualenv for EasyBuild (idempotent)
if [ ! -d "${EASYBUILD_VENV}" ]; then
  python3 -m venv "${EASYBUILD_VENV}"
fi

# EasyBuild sometimes expects "python" (not only "python3")
if [ ! -x "${EASYBUILD_VENV}/bin/python" ] && [ -x "${EASYBUILD_VENV}/bin/python3" ]; then
  ln -sf "${EASYBUILD_VENV}/bin/python3" "${EASYBUILD_VENV}/bin/python"
fi

# Install/upgrade EasyBuild inside the venv (idempotent)
"${EASYBUILD_VENV}/bin/pip" install -U pip setuptools wheel
"${EASYBUILD_VENV}/bin/pip" install -U easybuild easybuild-easyconfigs easybuild-easyblocks

# Clone the easyconfigs repository (optional but useful for --robot)
if [ ! -d "${EASYBUILD_EASYCONFIGS_REPO}/.git" ]; then
  rm -rf "${EASYBUILD_EASYCONFIGS_REPO}" 2>/dev/null || true
  git clone --depth 1 https://github.com/easybuilders/easybuild-easyconfigs.git "${EASYBUILD_EASYCONFIGS_REPO}"
else
  (cd "${EASYBUILD_EASYCONFIGS_REPO}" && git pull --ff-only) || true
fi

# Create an EasyBuild config file (idempotent)
mkdir -p "$(dirname "${EASYBUILD_CFG}")"
cat > "${EASYBUILD_CFG}" <<EOF
[config]
installpath = ${EASYBUILD_PREFIX}
installpath-software = ${EASYBUILD_PREFIX}/software
installpath-modules  = ${EASYBUILD_PREFIX}/modules

sourcepath = ${EASYBUILD_PREFIX}/sources
buildpath = ${EASYBUILD_PREFIX}/build
robot = ${EASYBUILD_EASYCONFIGS_REPO}
repositorypath = ${EASYBUILD_PREFIX}/ebfiles_repo

modules-tool = Lmod
module-naming-scheme = EasyBuildMNS

# Keep build logs (escape % as %%)
logfile-format = ${EASYBUILD_PREFIX}/logs,easybuild-%%(name)s-%%(version)s-%%(date)s.%%(time)s.log
EOF

# Ensure required subdirectories exist
mkdir -p "${EASYBUILD_PREFIX}"/{sources,build,software,modules,ebfiles_repo,logs}

# Add environment setup to ~/.bashrc (idempotent)
BASHRC_FILE="${HOME}/.bashrc"
if ! grep -q '>>> EASYBUILD >>>' "${BASHRC_FILE}" 2>/dev/null; then
cat >> "${BASHRC_FILE}" <<EOF

# >>> EASYBUILD >>>
# EasyBuild installation prefix
export EASYBUILD_PREFIX="${MY_SOFTWARE_INST_DIR}/easybuild"
export EASYBUILD_CONFIGFILES="\${EASYBUILD_PREFIX}/config.cfg"

# Make EasyBuild available (venv)
if [ -x "\${EASYBUILD_PREFIX}/venv/bin/eb" ]; then
  export PATH="\${EASYBUILD_PREFIX}/venv/bin:\${PATH}"
fi

# Initialize Lmod (modules) if available, and add EasyBuild module path
if [ -r /usr/share/lmod/lmod/init/bash ]; then
  source /usr/share/lmod/lmod/init/bash
  module use "\${EASYBUILD_PREFIX}/modules/all" 2>/dev/null || true
fi
# <<< EASYBUILD <<<
EOF

fi

# Show version if available
if [ -x "${EASYBUILD_VENV}/bin/eb" ]; then
  PATH="${EASYBUILD_VENV}/bin:${PATH}" "${EASYBUILD_VENV}/bin/eb" --version || true
fi

## Add paths to MODULEPATH for EasyBuild modules and non-Easybuild modules to /etc/profile (system-wide)
## Do it only if the lines are not already present in /etc/profile
#grep -qxF "module use ${MY_SOFTWARE_INST_DIR}/easybuild/modules/all" /etc/profile || echo "module use ${MY_SOFTWARE_INST_DIR}/easybuild/modules/all" | sudo tee -a /etc/profile > /dev/null
#grep -qxF "module use ${MY_NON_EB_MOD_DIR}" /etc/profile || echo "module use ${MY_NON_EB_MOD_DIR}" | sudo tee -a /etc/profile > /dev/null

# System-wide: make module paths available for all users (robust + idempotent)
EB_PROFILED_FILE="/etc/profile.d/easybuild-modulepaths.sh"

sudo tee "${EB_PROFILED_FILE}" > /dev/null <<EOF
# Ensure Lmod is initialized (if present), then add module paths.

if [ -r /usr/share/lmod/lmod/init/profile ]; then
  . /usr/share/lmod/lmod/init/profile
elif [ -r /etc/profile.d/lmod.sh ]; then
  . /etc/profile.d/lmod.sh
fi

if command -v module >/dev/null 2>&1; then
  module use "${MY_SOFTWARE_INST_DIR}/easybuild/modules/all" >/dev/null 2>&1 || true
  module use "${MY_NON_EB_MOD_DIR}" >/dev/null 2>&1 || true
fi
EOF

sudo chmod 0644 "${EB_PROFILED_FILE}"


# ---------------------------------------------------------------------
# Ubuntu 24.04 uses an externally managed system Python environment, so installing
# Python applications with "python3 -m pip install --user ..." is intentionally
# blocked. Use pipx instead: it creates and manages an isolated Python virtual
# environment for each command-line application and places the executable in
# ~/.local/bin.
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Installing pipx and then using pipx install fprettity..." 
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
sudo apt-get install -y pipx
pipx install fprettify


# Install fortls (Fortran Language Server) for VS Code / Modern Fortran
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
echo "Installing fortls Fortran Language Server using pipx..."
echo "━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━"
# Make sure ~/.local/bin exists and is on PATH for future interactive shells.
# pipx ensurepath is idempotent, but we also add an explicit ~/.bashrc entry
# below so that VS Code/WSL shells can find fortls reliably.
/usr/bin/pipx ensurepath || true
mkdir -p "${HOME}/.local/bin"

if ! grep -q '>>> LOCAL USER BIN >>>' "${HOME}/.bashrc" 2>/dev/null; then
cat >> "${HOME}/.bashrc" <<'EOF_BASHRC'

# >>> LOCAL USER BIN >>>
# User-level executables installed by tools such as pipx.
if [ -d "$HOME/.local/bin" ]; then
  export PATH="$HOME/.local/bin:$PATH"
fi
# <<< LOCAL USER BIN <<<
EOF_BASHRC
fi

# Install fortls if missing; otherwise upgrade the existing pipx-managed install.
if /usr/bin/pipx list 2>/dev/null | grep -q '^\s*package fortls\b'; then
  /usr/bin/pipx upgrade fortls
else
  /usr/bin/pipx install fortls
fi

# Show the installed fortls executable and version. Use the direct path because
# the updated ~/.local/bin PATH entry will take effect in newly opened shells.
if [ -x "${HOME}/.local/bin/fortls" ]; then
  "${HOME}/.local/bin/fortls" --version || true
else
  echo "WARNING: fortls was not found at ${HOME}/.local/bin/fortls."
  echo "Open a new WSL shell and check: which fortls"
fi
